\posterbox[postersection]{name=background, column=1, between=row3 and row7}{
  \sectiontitle{Background}
  \bodyfont
  \subhead{Minecraft resource-pack transfer}
  \begin{itemize}
    \item Resource packs contain hundreds of small tileable block faces.
    \item Manual pack creation is repetitive artistic work.
    \item Pixel-scale hue and edge errors are immediately visible.
  \end{itemize}
  \vspace{3pt}
  \subhead{Problem formulation}
  Given vanilla content texture \(x_c\), target pack \(p\), and style references \(S_p\), generate \(y\) that preserves material identity, adopts the target palette, and remains tileable.
}

  \posterfig{0.80}{figures/spritecraft-pipeline-overview.png}{Vanilla packs and target packs are filtered to valid full-cube textures, normalized, paired by filename, and exported as per-pack datasets.}
  \bodyfont
  \textbf{Scale:} 32$\times$32 RGB textures; three ranked style references; 20 validation textures; four fully trained \texttt{run33} target packs.

\posterbox[postersection]{name=method, column=1, between=row13 and row22}{
  \sectiontitle{Model}
  \bodyfont
  \subhead{Dual-model architecture}
  \begin{itemize}
    \item \textbf{StyleAwareUNet} (\(\sim\)7.7M parameters): denoising diffusion model conditioned on content, references, and timestep.
    \item \textbf{RecolorNet} (\(\sim\)1.0M parameters): deterministic residual U-Net for structure-preserving color transfer.
    \item Shared reference ranking uses texture descriptors plus filename priors.
  \end{itemize}
  \vspace{2pt}
  \posterfig{0.82}{figures/spritecraft-dual-model-architecture.png}{Diffusion provides expressive candidates; RecolorNet supplies a safer structural fallback.}
}

\posterbox[postersection]{name=training, column=1, between=row22 and row29}{
  \sectiontitle{Training Objective}
  \bodyfont
  \subhead{SNR-weighted diffusion supervision}
  \[
    \mathcal{L}_{total}=\mathcal{L}_{noise}
    + \bar{\alpha}_t(0.70\mathcal{L}_{recon}
    +0.30\mathcal{L}_{gradient}
    +0.20\mathcal{L}_{content})
  \]
  \subhead{Content-aware terms}
  \begin{itemize}
    \item Source-relative structure deltas preserve block identity.
    \item Target-gated contrast and hue focus on detailed regions.
    \item Color moments provide stable global palette gradients.
  \end{itemize}
}

\posterbox[postersection]{name=approach, column=2, between=row3 and row11}{
  \sectiontitle{Our Approach}
  \bodyfont
  \subhead{Output-driven routing}
  Instead of always trusting the generative branch, SpriteCraft scores the actual candidates and selects among diffusion, non-destructive detail injection, RecolorNet, and deterministic recolor fallback.
  \vspace{4pt}
  \begin{itemize}
    \item Structural quality checks edge fidelity, detail coherence, and entropy stability.
    \item Style quality compares the candidate with active support references.
    \item High-pattern textures such as \texttt{bookshelf} and \texttt{tnt\_side} use stricter gates.
  \end{itemize}
  \vspace{2pt}
  \posterfig{0.70}{figures/spritecraft-ensemble-routing-flow.png}{The selector favors plausible tile structure over target-aware MAE alone.}
}

\posterbox[postersection]{name=challenges, column=2, between=row24 and row29}{
  \sectiontitle{Challenges}
  \begin{minipage}[t]{0.49\linewidth}
    \bodyfont
    \subhead{Known failure modes}
    \begin{itemize}
      \item Transparent materials: alpha is not modeled.
      \item Fine symbols: TNT text and bookshelves need exact structure.
      \item High-resolution packs lose microtexture at 32x.
    \end{itemize}
    \subhead{Next steps}
    Alpha-aware training, learned support selection, multi-resolution refinement, and learned routing thresholds.
  \end{minipage}\hfill
  \begin{minipage}[t]{0.48\linewidth}
    \posterfig{0.95}{figures/run33-ashen-routing.png}{Recolor fallback protects fine structure when diffusion damages symbols or repeated narrow patterns.}
  \end{minipage}
}
